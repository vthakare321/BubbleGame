import { apiClient } from "@/api/client"; import { API_ENDPOINTS } from "@/api/endpoints"; import type { CreatePostDto, GetPostsParamsDto, PostDto, PostsResponseDto, UpdatePostDto, } from "../dto"; export const getPosts = async ( params?: GetPostsParamsDto ): Promise => { const { data } = await apiClient.get( API_ENDPOINTS.POSTS, { params, } ); return data; }; export const getPostById = async ( postId: number ): Promise => { const { data } = await apiClient.get( API_ENDPOINTS.POST_DETAILS(postId) ); return data; }; export const createPost = async ( payload: CreatePostDto ): Promise => { const { data } = await apiClient.post( API_ENDPOINTS.POSTS, payload ); return data; }; export const updatePost = async ( postId: number, payload: UpdatePostDto ): Promise => { const { data } = await apiClient.put( API_ENDPOINTS.POST_DETAILS(postId), payload ); return data; }; export const deletePost = async ( postId: number ): Promise => { await apiClient.delete(API_ENDPOINTS.POST_DETAILS(postId)); };
